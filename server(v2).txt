# anti rev-nuke system (by slanik)
num INFECTED_TEAM=2;         # team not allowed to nuke
num REVNUKE_KILL_DELAY=3000; # ms after detonation the jailed player dies
num NUKE_MAX_MS=20000;       # safety cap if nukeEnd never fires
str nukerID="";
bool nukeActive=false;
num nukeStartTime=0;
bool revPunishing=false;
str revPunishID="";
num revKillAt=0;

# teleport rev-nuker into the jail (rnX/rnY/rnZ) and mark for delayed kill
action startRevPunish(obj nuker) {
 nuker.position.x=rnX;nuker.position.y=rnY;nuker.position.z=rnZ;
 revPunishing=true;
 revPunishID=(str)nuker.id;
 revKillAt=GAME.TIME.now()+25000; # fallback, overridden on nukeEnd
 nukeActive=false;
 nukerID="";
 GAME.CHAT.broadcast((str)nuker.username+" tried to rev nuke but slanik was faster and caught him","#ff5555");
 logR("SYS","SERVER :: REV-NUKE JAIL :: "+(str)nuker.username);
}

# watch the active nuker + kill the jailed nuker after the delay (called from update)
action procRevWatch() {
 if(nukeActive){
  if(GAME.TIME.now()-nukeStartTime>NUKE_MAX_MS){nukeActive=false;nukerID="";}
  else if(nukerID!=""){
   obj nuker=fnByID(nukerID);
   if(notEmpty nuker){
    if(!(bool)nuker.active){nuker.respawn();}
    else if((num)nuker.team==INFECTED_TEAM){startRevPunish(nuker);}
   }
  }
 }
 if(revPunishing){
  if(GAME.TIME.now()>=revKillAt){
   obj rn=fnByID(revPunishID);
   if(notEmpty rn){(num)rn.health-=100000;}
   revPunishing=false;revPunishID="";
  }
 }
}

public action onCustomTrigger(str pId,str customParam,num value) {

 if(!notEmpty p){return;}
 if(customParam=="nukeStart"){
  if((num)p.team==INFECTED_TEAM){startRevPunish(p);}       # zombie hit the trigger -> instant jail
  else{nukerID=pId;nukeStartTime=GAME.TIME.now();nukeActive=true;} # human nuke -> watch them

  nukeActive=false;nukerID="";

 }
}

 logR("NET",(str)p.accountName+" :: sent valid <"+id+"> req :: SERVER");
 if(vld==1&&(str)data.r=="rq"){procDtReq(id,data,pID);}
